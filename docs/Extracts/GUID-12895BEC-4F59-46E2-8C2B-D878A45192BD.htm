Accessing Objects

 
 
 

The drawing dictionary Autodesk_PNP holds nonentity information such as LineGroups and Tag formats. The code below collects Style names into a StringCollection from the P&ID dictionary. 

//C#
public static void PnPGetStyles(out StringCollection styles)
{
   Database db = AcadApp.DocumentManager.MdiActiveDocument.Database;
   styles = new StringCollection();
   
   using (Transaction tr = db.TransactionManager.StartTransaction())
   {
      DBDictionary dictRoot = (DBDictionary)tr.GetObject(db.NamedObjectsDictionaryId, OpenMode.ForRead);
      DBDictionary dictPNP = (DBDictionary)tr.GetObject(dictRoot.GetAt("Autodesk_PNP"), OpenMode.ForRead);
      
      //Style Dictionary
      DBDictionary dictStyles = (DBDictionary)tr.GetObject(dictPNP.GetAt("PNP_STYLES"), OpenMode.ForRead);
      DbDictionaryEnumerator dictStylesEnum = dictStyles.GetEnumerator();
      while (dictStylesEnum.MoveNext())
         styles.Add(dictStylesEnum.Key);
   }
}
