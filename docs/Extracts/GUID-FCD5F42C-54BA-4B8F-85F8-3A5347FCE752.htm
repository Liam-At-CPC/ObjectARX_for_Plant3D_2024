Relationship Query (Sample)

 
 
 

The following code uses P3dLineGroupPartRelationship (Piping) or PipeLineGroupRelationship lists (P&ID) to get a list of all objects that belong to the group. 

public void ListLineDrawings() // This method can have any name
{
    //Project
    Editor ed = Application.DocumentManager.MdiActiveDocument.Editor;
    string dwgtype = PnPProjectUtils.GetActiveDocumentType();
    if (dwgtype != "PnId" && dwgtype != "Piping")
    {  //"PnId", "Piping", "Ortho", "Iso"
        ed.WriteMessage("This drawing is not a P&ID or 3D Model in the current project.\n");
        return;
    }
    Project prj = PnPProjectUtils.GetProjectPartForCurrentDocument();

    //Drawing and Data links
    DataLinksManager dlm = prj.DataLinksManager;
    PnPDatabase db = dlm.GetPnPDatabase();
    PnPTable tblDwg = db.Tables["PnPDrawings"];
    PnPTable tblDl = db.Tables["PnPDataLinks"];

    //Line Group
    string sLineGroup = "PipeLineGroup";
    if (dwgtype == "Piping") sLineGroup = "P3dLineGroup";
    PnPTable ctab = db.Tables[sLineGroup];
    PnPRow[] rowsLg = ctab.Select();

    foreach (PnPRow rowLg in rowsLg)
    {   
        //For each line gruop, use line group relationships to acquire the components in the line group.
        PnPRowIdArray idlgComponents = null;
        if (dwgtype == "Piping")
            //The PnPRelationshipTypes and PnPRoleTypes tables provide the string constants used below.
            idlgComponents = dlm.GetRelatedRowIds("P3dLineGroupPartRelationship", "LineGroup", rowLg.RowId, "Part");
        else //PnID
            //For P&ID drawings, you can also acquire data links directly from PipeLineGroup.  Here we illustrate relationships.
            idlgComponents = dlm.GetRelatedRowIds("PipeLineGroupRelationship", "PipeLineGroup", rowLg.RowId, "PipeLine");

        //Get the line number, a property of the line group.
        string sLineCol="LineNumber";
        if (dwgtype == "Piping") sLineCol = "Number";
        string sLine = getRowVal(rowLg, sLineCol).ToString();
        ed.WriteMessage("\nLine number " + sLine + " contains:");
        foreach (int ridComponent in idlgComponents) 
        {
            //List component type and drawing for each line group item.
            PnPRow rowComponent = db.GetRow(ridComponent);
            PnPRow[] rowsDl = tblDl.Select(string.Format("RowId={0}", ridComponent));
            if (rowsDl.Length != 1) continue; //Error
            object sDwg = getRowVal(rowsDl[0], "DwgId");
            PnPRow[] rowsDwg = tblDwg.Select(string.Format("PnPID={0}", (int)sDwg));
            if (rowsDwg.Length != 1) continue; //Error
            string sPath = getRowVal(rowsDwg[0], "Path").ToString();
            ed.WriteMessage("\n   " + rowComponent.ClassName + " in " + sPath);
        }
    }
}
