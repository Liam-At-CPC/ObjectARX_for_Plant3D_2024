Project Settings and Configuration

 
 
 

The following code lists a sample of Project settings to the command line. 

//C#
{
   Project oPnPActiveProject = PnPActiveProject();
   if (oPnPActiveProject == null) return;
   
   Editor oEditor = AcadApp.DocumentManager.MdiActiveDocument.Editor;
   
   oEditor.WriteMessage("\nProjectName: " + oProject.ProjectName);
   oEditor.WriteMessage("\n   AssembliesDirectory: " + oProject.AssembliesDirectory);
   oEditor.WriteMessage("\n   ProjectDescription: " + oProject.ProjectDescription);
   oEditor.WriteMessage("\n   ProjectDirectory: " + oProject.ProjectDirectory);
   oEditor.WriteMessage("\n   ProjectDwgDirectory: " + oProject.ProjectDwgDirectory);
   oEditor.WriteMessage("\n   ReadOnly: " + oProject.ReadOnly);
   oEditor.WriteMessage("\n   SymbolsDirectory: " + oProject.SymbolsDirectory);
   oEditor.WriteMessage("\n   TypicalsDirectory: " + oProject.TypicalsDirectory);
   oEditor.WriteMessage("\n   Version: " + oProject.Version);
}
